// Pattern for Modal
function createBootstrapModal(modalEl, options = {}) {
    let modal = null;
    
    if (typeof bootstrap !== 'undefined' && bootstrap && bootstrap.Modal) {
        try {
            // Try Bootstrap 5.1+ getOrCreateInstance
            if (bootstrap.Modal.getOrCreateInstance && 
                typeof bootstrap.Modal.getOrCreateInstance === 'function') {
                modal = bootstrap.Modal.getOrCreateInstance(modalEl);
            }
            // Try Bootstrap 5.0 getInstance
            else if (bootstrap.Modal.getInstance && 
                     typeof bootstrap.Modal.getInstance === 'function') {
                modal = bootstrap.Modal.getInstance(modalEl);
                if (!modal && typeof bootstrap.Modal === 'function') {
                    modal = new bootstrap.Modal(modalEl, options);
                }
            }
            // Try direct constructor
            else if (typeof bootstrap.Modal === 'function') {
                modal = new bootstrap.Modal(modalEl, options);
            }
        } catch (e) {
            console.warn('Bootstrap Modal API failed, falling back to jQuery:', e);
        }
    }
    
    // Fallback to jQuery/Bootstrap 4
    if (!modal) {
        return {
            show: function() { $(modalEl).modal('show'); },
            hide: function() { $(modalEl).modal('hide'); },
            dispose: function() { $(modalEl).modal('dispose'); }
        };
    }
    
    return modal;
}

// Pattern for Tooltip
function createBootstrapTooltip(element, options = {}) {
    if (typeof bootstrap !== 'undefined' && bootstrap && bootstrap.Tooltip) {
        try {
            if (typeof bootstrap.Tooltip === 'function') {
                return new bootstrap.Tooltip(element, options);
            }
        } catch (e) {
            console.warn('Bootstrap Tooltip failed:', e);
        }
    }
    // Fallback: Return null or use jQuery tooltip
    return null;
}
